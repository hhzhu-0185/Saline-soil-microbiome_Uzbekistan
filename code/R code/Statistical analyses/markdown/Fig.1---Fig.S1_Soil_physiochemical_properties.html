### 1. pH

rm(list=ls())
setwd("E:/Saline-alkali soil/Saline-alkali/amplicon/result-U/AMF/otu")

# read metadata file
group <- read.csv("sample_metadata.csv")
group <- na.omit(group)
group$cnratio <- group$tc/group$tn
group$npratio <- group$tn/group$tp
group$cpratio <- group$tc/group$tp
group$salt_content <- group$salt_content/10

group2 <- group[group$country == "Uzbekistan",]

# Calculating mean and stand stand deviation by plants
mean<-aggregate(pH~plants, data = group2, FUN = "mean")
names(mean)[2] <- "mean"
sd<-aggregate(pH~plants, data = group2, FUN = "sd")
names(sd)[2] <- "sd"

# Combining mean,stand deviation with property values
# Merging the three elements continuously with Reduce
group2 <- Reduce(function(x, y) merge(x, y, by = "plants", all = TRUE), list(group2, mean, sd))


# Nonparametric test
ST <- with(group2,kruskal(pH, plants, p.adj = "bonferroni"))

# Extracting Chi square value and P value from the result
statis <- as.data.frame(rbind(ST$statistics[ ,1:3]))

# Adding significance
statis$sig <- 0
statis$sig = ifelse(statis$p.chisq < 0.001,
                    rep(" < 0.001", nrow(statis[statis$p.chisq < 0.001, ])),
                    round(statis[statis$p.chisq > 0.001, ]$p.chisq, 3))

# Creating dataframe for Chi square value and P value
# Sprintf for pasting texts and values
# asterisk (*) among elements represent pasting without space, tilde (~) among elements represent pasting witht space
rp.value <- ifelse(statis$p.chisq < 0.001, 
                   sprintf("italic(P) == %.2f %%*%% 10^%.0f ~ (italic(chi^2) == %.3g)", 
                           statis$p.chisq / 10^floor(log10(statis$p.chisq)), 
                           floor(log10(statis$p.chisq)), 
                           statis$Chisq),
                   sprintf("italic(P) == %.3g ~ (italic(chi^2) == %.3g)", statis$p.chisq, statis$Chisq))


# Extracting multiple comparision result
statis1 <- as.data.frame(rbind(ST$groups))
# Extracting mean of Nonparametric tests
statis2 <- as.data.frame(rbind(ST$means))

# Extracting Ymax for add letters of multiple comparision
p1 <- ggplot(data = group2, aes(y = pH,x = factor(plants))) +
      stat_boxplot(geom = "errorbar",  width = 0.3)
ymaxdata <- layer_data(p1)
levels(factor(group2$plants)) == rownames(statis2)  # If the order of Ymax is the same as the order of mean of Nonparametric tests
## [1] TRUE TRUE TRUE TRUE TRUE
statis2$ymax <- ymaxdata$ymax

# Merging dataframe of multiple comparision result with dataframe of mean of nonparametric tests
statis1$id <- rownames(statis1)
statis2$id <- rownames(statis2)
statis2 <- merge(statis2, statis1, by = "id")

# Creating dataframe of geom_text for letters of nonparametric tests
multcomp.data <- data.frame(abc=factor(statis2$groups),
                            x = statis2$id,
                            y = as.numeric(statis2$ymax))

# plot
group2$plants <- factor(group2$plants, levels = c("Bare land", "Suaeda", "Tamarix chinensis", "Cotton", "Maize"))
p_ph <- ggplot(data = group2, aes(y = pH, x = plants, color= plants)) +
        stat_boxplot(geom = "errorbar", width = 0.3) +
        geom_boxplot(width = 0.5) +
        #  geom_jitter(size=0.8) +
        labs(title = "pH") +
        scale_x_discrete(labels = c(
                                    'Bare land' = "Bareland", 
                                    'Cotton' = "Cotton", 
                                    'Maize' = "Maize", 
                                    'Suaeda' = "Suaeda",
                                    'Tamarix chinensis' = "Tamarix")) +
        scale_color_manual(name = "Habitat", 
                           values = c("red","blue","purple","darkgreen","deepskyblue"),
                           labels = c( "Bare land"="Bareland", 
                                       "Suaeda"="Suaeda",
                                       "Tamarix chinensis"="Tamarix",
                                       "Cotton", 
                                       "Maize")) + 
        annotate("text",x=3, y=9.7, label=rp.value, col = "red", size = 5, parse = TRUE) + 
        geom_text(data = multcomp.data,aes(x = x,y = y + 0.25,label = abc), color ="black", size = 6, hjust = 0.5) +
        theme_bw() +
        ylim(7,9.8) +
        theme(panel.grid = element_blank(), 
              axis.ticks.y = element_line(color = "black", linewidth = 0.5)) +
        theme(plot.title = element_text(hjust = 0.5, size = 20, face = "bold"),
              
              axis.title = element_blank(),
              axis.text = element_text(size = 12, face = "bold", color = "black"),
              legend.title = element_text(size = 16, face = "bold", color = "black", hjust=0),
              legend.text = element_text(size = 12, face = "bold", color = "black"))
